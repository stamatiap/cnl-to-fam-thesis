
(* ---- Top-level structure ---- *)

attack          ::= header background event+ detection?

header          ::= tactics technique

background      ::= "Background" "Given" assets

assets          ::= asset_definition ("And" asset_definition)*

asset_definition ::= asset_type "as" asset_name


(* ---- Header components ---- *)

tactics         ::= "Tactics" ":" tactic ("," tactic)*

tactic          ::= TACTIC_ID identifier

technique       ::= "Technique" ":" TECHNIQUE_ID identifier


(* ---- Event blocks ---- *)

event           ::= "Event" digit
                    given_clause
                    when_clause
                    then_clause
                    timing?

given_clause    ::= "Given" given_item ( ("And" given_item)*
                                       | ("Or"  given_item)*
                                       | ("Xor" given_item)* )

given_item      ::= event_ref | state_condition

event_ref       ::= "Event" digit

when_clause     ::= "When" action

action          ::= actor action_verb action_object modifier*

then_clause     ::= "Then" state_condition ("And" state_condition)*

state_condition ::= condition_object state_verb modifier*

state_verb      ::= "is" identifier
                  | identifier


(* ---- Modifiers ---- *)

modifier        ::= location | destination | source | geolocation

location        ::= "in" asset_name
destination     ::= "to" asset_name
source          ::= ("by" | "from") asset_name
geolocation     ::= "located_at" geo_location
content         ::= "CONTAINING" content_value
timing          ::= "During" time_window


(* ---- Detection block ---- *)

detection       ::= "Detection" detection_expr

detection_expr  ::= event_ref ( ("And" | "Or" | "Xor") event_ref )*


(* ---- Terminals ---- *)

asset_type      ::= identifier
asset_name      ::= identifier
actor           ::= asset_name
action_object   ::= asset_name
condition_object ::= asset_name
action_verb     ::= identifier
time_window     ::= identifier
geo_location    ::= identifier
content_value   ::= identifier

TACTIC_ID       ::= "TA" digit{4}
TECHNIQUE_ID    ::= "T" digit{4} ("." digit{3})*
identifier      ::= letter (letter | digit | "_")*
digit           ::= [0-9]+
